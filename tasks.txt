# Test task list. One shell command per line.
# Blank lines and lines starting with '#' are ignored.
#
# The naive runner only handles the "happy path" tasks (1-3) well.
# Tasks 4-7 are the troublesome ones the exercise warns about; they are
# listed here so the production version can be tested against them, but
# running them with the naive runner will hang or misbehave.

# --- happy path: short tasks that succeed quickly ---
echo "hello from task A"
python3 -c "print(sum(range(1000)))"
sh -c "for i in 1 2 3; do echo tick $i; done"

# --- trouble: runs far too long (should be timed out & killed) ---
sleep 600

# --- trouble: tries to eat all memory (should be OOM-killed, not host death) ---
python3 -c "b=bytearray(); \nwhile True: b.extend(bytearray(10*1024*1024))"

# --- trouble: forks a swarm of child processes (all must die on kill) ---
sh -c "for i in $(seq 1 20); do sleep 600 & done; wait"

# --- trouble: fails on purpose (non-zero exit; should be classified failed) ---
sh -c "echo boom >&2; exit 7"
